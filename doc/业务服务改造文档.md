# 业务服务改造文档

## 1. 改造概述

### 1.1 改造目标
将现有 `va_visionai_server` 中的AIGC功能（图片/视频生成、LLM聊天）剥离至独立的AIGC基础服务，通过统一的客户端调用实现功能解耦。

### 1.2 改造范围
**需要改造的模块：**
- **异步任务模块**：
  - `internal/api/picture_forge.go` - 图片视频生成API
  - `internal/service/picture_generate/` - 图片生成服务
  - `internal/task/picture_workflow/` - 后台处理器
- **聊天模块**：
  - `internal/api/chat.go` - 聊天API  
  - `internal/service/chat/` - 聊天服务
- **公共模块**：
  - `cmd/main.go` - 应用启动逻辑
  - `internal/bootstrap/` - 依赖注入

**保持不变的模块：**
- 用户认证、授权、计费管理
- 社区、分享等业务功能

### 1.3 改造策略
- 采用**特性开关**控制新旧逻辑切换
- **渐进式迁移**，确保零停机时间
- **向后兼容**，随时可回滚

## 2. AIGC客户端设计

### 2.1 接口定义
客户端需要同时支持异步任务（图片/视频生成）和实时流式（LLM聊天）两种模式。

### 2.2 客户端实现
客户端封装所有与AIGC服务的交互，处理连接管理、认证、重试等。

```go
// internal/clients/aigc/client.go
type Client struct {
    conn         *grpc.ClientConn
    client       aigc_v1.AIGCServiceClient
    config       *Config
    metrics      *ClientMetrics
    retryPolicy  *RetryPolicy
}

type Config struct {
    ServerAddr      string        `yaml:"server_addr"`
    APIKey          string        `yaml:"api_key"`
    Timeout         time.Duration `yaml:"timeout"`
    MaxRetries      int           `yaml:"max_retries"`
    RetryInterval   time.Duration `yaml:"retry_interval"`
    KeepAlive       bool          `yaml:"keep_alive"`
    ConnPoolSize    int           `yaml:"conn_pool_size"`
}

type RetryPolicy struct {
    MaxRetries    int
    InitialDelay  time.Duration
    MaxDelay      time.Duration
    Multiplier    float64
}

func NewClient(config *Config) (*Client, error) {
    // gRPC连接选项
    opts := []grpc.DialOption{
        grpc.WithInsecure(),
        grpc.WithTimeout(config.Timeout),
        grpc.WithDefaultCallOptions(
            grpc.MaxCallRecvMsgSize(32*1024*1024), // 32MB
            grpc.MaxCallSendMsgSize(32*1024*1024),
        ),
    }
    
    if config.KeepAlive {
        opts = append(opts, grpc.WithKeepaliveParams(keepalive.ClientParameters{
            Time:                10 * time.Second,
            Timeout:             3 * time.Second,
            PermitWithoutStream: true,
        }))
    }
    
    conn, err := grpc.Dial(config.ServerAddr, opts...)
    if err != nil {
        return nil, fmt.Errorf("failed to connect to AIGC service: %w", err)
    }
    
    return &Client{
        conn:   conn,
        client: aigc_v1.NewAIGCServiceClient(conn),
        config: config,
        metrics: NewClientMetrics(),
        retryPolicy: &RetryPolicy{
            MaxRetries:   config.MaxRetries,
            InitialDelay: config.RetryInterval,
            MaxDelay:     30 * time.Second,
            Multiplier:   2.0,
        },
    }, nil
}

func (c *Client) SubmitAsyncTask(ctx context.Context, req *AsyncTaskRequest) (*aigc_v1.SubmitAsyncTaskResponse, error) {
    start := time.Now()
    defer func() {
        c.metrics.RecordCall("SubmitAsyncTask", time.Since(start))
    }()
    
    grpcReq := &aigc_v1.SubmitAsyncTaskRequest{
        TaskType:     req.TaskType,
        WorkflowName: req.WorkflowName,
        Params:       req.Params,
        CallbackUrl:  req.CallbackURL,
        Metadata:     req.Metadata,
    }
    
    var resp *aigc_v1.SubmitAsyncTaskResponse
    var err error
    
    err = c.retryWithBackoff(ctx, func() error {
        ctx = c.withAuth(ctx)
        resp, err = c.client.SubmitAsyncTask(ctx, grpcReq)
        return err
    })
    
    if err != nil {
        c.metrics.RecordError("SubmitAsyncTask")
        return nil, fmt.Errorf("submit async task failed: %w", err)
    }
    
    return resp, nil
}

func (c *Client) GetAsyncTask(ctx context.Context, taskID string) (*aigc_v1.GetAsyncTaskResponse, error) {
    start := time.Now()
    defer func() {
        c.metrics.RecordCall("GetAsyncTask", time.Since(start))
    }()
    
    grpcReq := &aigc_v1.GetAsyncTaskRequest{TaskId: taskID}
    
    var resp *aigc_v1.GetAsyncTaskResponse
    var err error
    
    err = c.retryWithBackoff(ctx, func() error {
        ctx = c.withAuth(ctx)
        resp, err = c.client.GetAsyncTask(ctx, grpcReq)
        return err
    })
    
    if err != nil {
        c.metrics.RecordError("GetAsyncTask")
        return nil, fmt.Errorf("get async task failed: %w", err)
    }
    
    return resp, nil
}

func (c *Client) GetAsyncTasks(ctx context.Context, taskIDs []string) ([]*aigc_v1.GetAsyncTaskResponse, error) {
    results := make([]*aigc_v1.GetAsyncTaskResponse, 0, len(taskIDs))
    
    // 并发查询多个任务，提高效率
    type result struct {
        resp *aigc_v1.GetAsyncTaskResponse
        err  error
        idx  int
    }
    
    resultChan := make(chan result, len(taskIDs))
    sem := make(chan struct{}, 10) // 限制并发度为10
    
    for i, taskID := range taskIDs {
        go func(idx int, id string) {
            sem <- struct{}{}
            defer func() { <-sem }()
            
            resp, err := c.GetAsyncTask(ctx, id)
            resultChan <- result{resp: resp, err: err, idx: idx}
        }(i, taskID)
    }
    
    // 收集结果
    resultMap := make(map[int]*aigc_v1.GetAsyncTaskResponse)
    var firstError error
    
    for i := 0; i < len(taskIDs); i++ {
        res := <-resultChan
        if res.err != nil && firstError == nil {
            firstError = res.err
        }
        if res.resp != nil {
            resultMap[res.idx] = res.resp
        }
    }
    
    // 按原顺序返回结果
    for i := 0; i < len(taskIDs); i++ {
        if resp, ok := resultMap[i]; ok {
            results = append(results, resp)
        } else {
            results = append(results, nil)
        }
    }
    
    return results, firstError
}

func (c *Client) ChatStream(ctx context.Context) (aigc_v1.AIGCService_ChatStreamClient, error) {
    start := time.Now()
    defer func() {
        c.metrics.RecordCall("ChatStream", time.Since(start))
    }()
    
    ctx = c.withAuth(ctx)
    stream, err := c.client.ChatStream(ctx)
    if err != nil {
        c.metrics.RecordError("ChatStream")
        return nil, fmt.Errorf("create chat stream failed: %w", err)
    }
    
    return stream, nil
}

func (c *Client) HealthCheck(ctx context.Context) error {
    ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()
    
    // 简单的健康检查：尝试建立连接
    conn := c.conn
    state := conn.GetState()
    
    if state == connectivity.Ready || state == connectivity.Idle {
        return nil
    }
    
    return fmt.Errorf("AIGC service unhealthy, connection state: %v", state)
}

func (c *Client) withAuth(ctx context.Context) context.Context {
    return metadata.AppendToOutgoingContext(ctx, "x-api-key", c.config.APIKey)
}

func (c *Client) retryWithBackoff(ctx context.Context, fn func() error) error {
    var lastErr error
    delay := c.retryPolicy.InitialDelay
    
    for i := 0; i <= c.retryPolicy.MaxRetries; i++ {
        if i > 0 {
            select {
            case <-time.After(delay):
            case <-ctx.Done():
                return ctx.Err()
            }
            
            delay = time.Duration(float64(delay) * c.retryPolicy.Multiplier)
            if delay > c.retryPolicy.MaxDelay {
                delay = c.retryPolicy.MaxDelay
            }
        }
        
        if err := fn(); err != nil {
            lastErr = err
            
            // 检查是否可重试
            if !c.isRetryableError(err) {
                return err
            }
            continue
        }
        
        return nil
    }
    
    return fmt.Errorf("max retries exceeded, last error: %w", lastErr)
}

func (c *Client) isRetryableError(err error) bool {
    if err == nil {
        return false
    }
    
    // gRPC错误码检查
    if s, ok := status.FromError(err); ok {
        switch s.Code() {
        case codes.Unavailable, codes.DeadlineExceeded, codes.ResourceExhausted:
            return true
        }
    }
    
    // 网络错误检查
    if strings.Contains(err.Error(), "connection refused") ||
       strings.Contains(err.Error(), "timeout") {
        return true
    }
    
    return false
}

func (c *Client) Close() error {
    if c.conn != nil {
        return c.conn.Close()
    }
    return nil
}

// internal/api/picture_forge.go
type PictureForgeServer struct {
    vai.UnimplementedPictureForgeServiceServer

func (s *PictureForgeServer) SubmitPictureForgeTask(ctx context.Context, req *vai.SubmitPictureForgeTaskRequest) (*vai.SubmitPictureForgeTaskResponse, error) {
    // 1. 前置检查：用户权限、额度等
    if err := s.validateRequest(ctx, req); err != nil {
        return s.buildErrorResponse(err)
    }
    
    // 2. 根据特性开关选择处理路径
    if s.useAIGCService {
        return s.submitViaAIGC(ctx, req)
    }
    return s.submitViaLegacy(ctx, req)
}

func (s *PictureForgeServer) submitViaAIGC(ctx context.Context, req *vai.SubmitPictureForgeTaskRequest) (*vai.SubmitPictureForgeTaskResponse, error) {
    // 1. 创建业务任务记录
    businessTask, err := s.createBusinessTask(ctx, req)
    if err != nil {
        return nil, err
    }
    
    // 2. 转换参数并调用AIGC服务
    aigcReq := &aigc.AsyncTaskRequest{
        TaskType:     s.mapTaskType(req.GetWorkflowId()),
        WorkflowName: req.GetWorkflowId(),
        Params:       s.convertParams(req),
        CallbackURL:  s.buildCallbackURL(),
        Metadata: map[string]string{
            "business_task_id": businessTask.TaskID,
            "user_id":          req.GetRequestHeader().GetUserId(),
        },
    }
    
    aigcResp, err := s.aigcClient.SubmitAsyncTask(ctx, aigcReq)
    if err != nil {
        s.markTaskFailed(ctx, businessTask.TaskID, err.Error())
        return nil, err
    }
    
    // 3. 关联AIGC任务ID
    s.updateTaskWithAIGCID(ctx, businessTask.TaskID, aigcResp.TaskId)
    
    // 4. 扣费处理
    s.deductUserAmount(ctx, req)
    
    return &vai.SubmitPictureForgeTaskResponse{
        TaskId: businessTask.TaskID,
    }, nil
}

// internal/api/chat.go
type ChatServer struct {
    vai.UnimplementedChatServiceServer
    
    // 保留原有依赖
    chatService   *service.ChatService
    llmService    *service.LLMService
    
    // 新增AIGC客户端
    aigcClient    aigc.ClientInterface
    useAIGCChat   bool // 特性开关
}

func (s *ChatServer) ChatStream(req *vai.ChatStreamRequest, stream vai.ChatService_ChatStreamServer) error {
    // 1. 前置检查
    if err := s.validateChatRequest(req); err != nil {
        return err
    }
    
    // 2. 根据特性开关选择处理路径
    if s.useAIGCChat {
        return s.chatViaAIGC(req, stream)
    }
    return s.chatViaLegacy(req, stream)
}

func (s *ChatServer) chatViaAIGC(req *vai.ChatStreamRequest, stream vai.ChatService_ChatStreamServer) error {
    ctx := stream.Context()
    
    // 1. 建立与AIGC服务的流连接
    aigcStream, err := s.aigcClient.ChatStream(ctx)
    if err != nil {
        return status.Errorf(codes.Internal, "连接AIGC服务失败: %v", err)
    }
    defer aigcStream.CloseSend()
    
    // 2. 发送请求到AIGC服务
    aigcReq := &aigc_v1.ChatStreamRequest{
        ModelName:  req.Model,
        Messages:   s.convertMessages(req.Messages),
        Parameters: s.convertParameters(req.Parameters),
        Metadata:   map[string]string{
            "user_id": req.GetRequestHeader().GetUserId(),
        },
    }
    
    if err := aigcStream.Send(aigcReq); err != nil {
        return status.Errorf(codes.Internal, "发送请求失败: %v", err)
    }
    
    // 3. 转发响应流
    for {
        aigcResp, err := aigcStream.Recv()
        if err == io.EOF {
            break
        }
        if err != nil {
            return status.Errorf(codes.Internal, "接收响应失败: %v", err)
        }
        
        // 转换为业务响应格式
        businessResp := s.convertChatResponse(aigcResp)
        if err := stream.Send(businessResp); err != nil {
            return err
        }
    }
    
    return nil
}
```

func (s *ChatServer) convertMessages(messages []*vai.ChatMessage) []*aigc_v1.ChatMessage {
    result := make([]*aigc_v1.ChatMessage, len(messages))
    for i, msg := range messages {
        result[i] = &aigc_v1.ChatMessage{
            Role:    msg.Role,
            Content: msg.Content,
        }
    }
    return result
}

func (s *ChatServer) convertChatResponse(aigcResp *aigc_v1.ChatStreamResponse) *vai.ChatStreamResponse {
    return &vai.ChatStreamResponse{
        ChunkId:      aigcResp.ChunkId,
        ContentChunk: aigcResp.ContentChunk,
        IsFinish:     aigcResp.IsFinish,
        FinishReason: aigcResp.FinishReason,
        Usage:        aigcResp.Usage,
    }
}
